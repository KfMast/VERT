```
public async convert(input: VertFile, to: string): Promise<VertFile> {
    if (!to.startsWith(".")) to = `.${to}`;
    const isAlac = to === ".alac";
    if (isAlac) to = ".m4a";

    let conversionError: string | null = null;
    const ffmpeg = await this.setupFFmpeg(input);
    this.activeConversions.set(input.id, ffmpeg);

    // 监听错误
    const errorListener = (l: { message: string }) => {
        // 错误处理逻辑
    };
    ffmpeg.on("log", errorListener);

    // 写入文件到虚拟文件系统
    const buf = new Uint8Array(await input.file.arrayBuffer());
    await ffmpeg.writeFile("input", buf);

    // 构建并执行转换命令
    const command = await this.buildConversionCommand(ffmpeg, input, 
    to, isAlac);
    await ffmpeg.exec(command);

    // 读取转换结果
    const output = (await ffmpeg.readFile("output" + to)) as unknown as 
    Uint8Array;
    
    // 清理资源
    ffmpeg.off("log", errorListener);
    ffmpeg.terminate();

    // 返回转换后的文件
    const outputFileName = input.name.split(".").slice(0, -1).join(".") 
    + to;
    const outBuf = new Uint8Array(output).buffer.slice(0);
    return new VertFile(new File([outBuf], outputFileName), to);
}
```

### 4. 编解码器选择
```
// 根据文件扩展名获取编解码器
const getCodecs = (
    ext: string,
    isAlac: boolean = false,
): { video: string; audio: string } => {
    switch (ext) {
        // 音频格式
        case ".mp3":
            return { video: "libx264", audio: "libmp3lame" };
        case ".flac":
            return { video: "libx264", audio: "flac" };
        case ".wav":
            return { video: "libx264", audio: "pcm_s16le" };
        case ".ogg":
        case ".oga":
            return { video: "libx264", audio: "libvorbis" };
        case ".opus":
            return { video: "libx264", audio: "libopus" };
        case ".aac":
            return { video: "libx264", audio: "aac" };
        case ".m4a":
            return {
                video: "libx264",
                audio: isAlac ? "alac" : "aac",
            };
        case ".alac":
            return { video: "libx264", audio: "alac" };
        case ".wma":
            return { video: "libx264", audio: "wmav2" };
        // 默认编解码器
        default:
            return { video: "libx264", audio: "aac" };
    }
};
```

## 转换流程详解
### 1. 初始化阶段
- 创建 FFmpeg 实例并加载 WebAssembly 核心文件
- 监听转换进度和错误信息
### 2. 文件处理阶段
- 将输入文件写入 FFmpeg 虚拟文件系统
- 检测文件格式和特性（如采样率、比特率）
### 3. 命令构建阶段
- 根据输入/输出格式确定转换类型（音频→音频、视频→音频）
- 构建 FFmpeg 转换命令，包含以下参数：
  - 输入文件路径
  - 音频编解码器选择
  - 比特率设置（根据用户配置或自动检测）
  - 采样率设置（根据用户配置或自动检测）
  - 元数据处理（保留或移除）
  - 输出文件路径
### 4. 执行转换阶段
- 调用 ffmpeg.exec() 执行转换命令
- 实时更新转换进度
- 捕获并处理转换过程中的错误
### 5. 结果处理阶段
- 从 FFmpeg 虚拟文件系统读取转换结果
- 创建新的 VertFile 对象包装转换后的文件
- 清理资源（终止 FFmpeg 实例，释放内存）
## 技术特点
### 1. 纯前端实现
- 使用 WebAssembly 版本的 FFmpeg，无需后端服务器
- 所有转换操作在浏览器中完成，保护用户隐私
### 2. 智能参数设置
- 比特率 ：支持用户自定义或自动检测输入文件比特率
- 采样率 ：支持用户自定义或自动检测输入文件采样率
- 元数据 ：可选择保留或移除音频元数据（如标题、艺术家等）
### 3. 特殊格式处理
- ALAC 格式 ：自动处理为 m4a 容器
- M4A 格式 ：支持保留专辑封面（作为视频流）
- Opus 格式 ：自动调整不支持的采样率（如将 44100Hz 调整为 48000Hz）
### 4. 错误处理
- 检测并处理不支持的采样率错误
- 检测无音频流错误
- 提供详细的错误提示信息
## 支持的转换类型
转换类型 描述 实现方式 音频 → 音频 不同音频格式之间的转换 直接使用 FFmpeg 命令转换 视频 → 音频 从视频中提取音频轨道 使用 -map 0:a:0 映射音频流 音频 → 视频 将音频转换为视频（使用专辑封面或纯色背景） 构建包含视频滤镜的命令

## 性能与优化
### 1. 性能考虑
- 内存限制 ：浏览器对 WebAssembly 内存使用有限制，可能无法处理过大的音频文件
- 执行速度 ：WebAssembly 执行速度低于原生代码，转换大型文件可能较慢
- CPU 使用率 ：转换过程可能占用大量 CPU 资源
### 2. 优化措施
- 独立实例 ：为每个转换任务创建独立的 FFmpeg 实例
- 资源清理 ：转换完成后及时终止 FFmpeg 实例，释放内存
- 进度监控 ：实时显示转换进度，提升用户体验
- 错误处理 ：完善的错误检测和提示机制
## 总结
VERT 项目的音频转换通过 WebAssembly 版本的 FFmpeg 实现纯前端处理，支持多种音频格式之间的转换，以及从视频中提取音频。核心流程包括初始化 FFmpeg、写入文件、构建命令、执行转换和处理结果，同时提供智能参数设置和错误处理机制，为用户提供便捷、安全的音频转换功能。
